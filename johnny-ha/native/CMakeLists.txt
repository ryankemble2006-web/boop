cmake_minimum_required(VERSION 3.22)
project(johnny C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_JOYSTICK OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(SDL_HIDAPI OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
add_subdirectory(${SDL_SOURCE} sdl)
set(ENGINE utils uncompress resource story walk calcpath ads ttm island bench graphics sound events config)
foreach(name IN LISTS ENGINE)
  list(APPEND ENGINE_FILES "${REBORN_SOURCE}/${name}.c")
endforeach()
add_library(johnny SHARED ha_port.c ${ENGINE_FILES})
target_include_directories(johnny PRIVATE . ${REBORN_SOURCE} ${SDL_SOURCE}/include)
target_link_libraries(johnny PRIVATE SDL2-static log m)
target_compile_options(johnny PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
target_link_options(johnny PRIVATE -Wl,-z,defs -Wl,-z,max-page-size=16384)
